 27def renderTemplate(template_file : str, **kwargs : Any) -> str:
 28    """Load templates from the current directory
 29
 30    Args:

 33

497                        else:
498                            mime_main, mime_subtype = 'application', 'octet-stream'

500                        msg.add_attachment(file_data, maintype=mime_main, subtype=mime_subtype, filename=file_name)
501                        print(f"Attached file: {file_name}")
502                except Exception as e:
503                    print(f"Failed to attach {file_path}: {e}")
504
505        # Adding ssl layer of security
506        ssl_context = ssl.create_default_context()
507
508        try:
509            # Creating smtp server
510            print("Connecting to Server....")
511            my_server = smtplib.SMTP(self.smtp_server, self.smtp_port)
512            my_server.starttls(context=ssl_context)
513
514            # Login to smtp server
515            my_server.login(self.sender_email, self.password)
516            print("Connected to server!")
517

525            print(f"....\nSuccessfully sent to: {self.receiver}")
526            print("********************************************")
527            del msg["To"]
528        except Exception as e:
529            print(f"ERROR: {e}")
530        finally:
531            my_server.quit()
